\chapter{In Vigilia Corporis Christi}

\lelem{Ad completorium super psalmos antiphona}
\lettrine{A}{} fructu frumenti et vini multiplicati:
fideles in pace Christi requiescunt.\quickref{can:corpus_christi_afructu}

\lelem{Capitulum}
\lettrine{Q}{ui} enim manducat et bibit indigne:
iudicium sibi manducat et bibit:
non diiudicans corpus Domini. D\editorial{eo gratias}.

\lelem{Hymnus}
\hymnus{verbum_caro}

\elem{V.}
Posuit fines tuos pacem.
Et adipe frumenti satiat te: alleluia.

\lelem{Ad Nunc dimittis antiphona}
\lettrine{V}{incenti} dabo manna absconditum: et nomen novum alleluia.\quickref{can:corpus_christi_vincenti}

\rubrica{Preces consuete\quickref{preces} dicuntur: adiuncto versiculo isto.}\\
Caro mea vere est cibus.
Et sanguis me\editorial{us} ve\editorial{rus} est po\editorial{tus}. alleluia.

\lelem{Oratio}
\lettrine{F}{ac} nos quaesumus Domine
divinitatis tuae sempiterna fruitione repleri:
quam preciosi corporis et sanguinis tui:
t\editorial{empor}alis perceptio prefigurat.
Qui cum Deo Patre.

\rubrica{Hic ordo Completorii non mutatur per totam hanc octavam
  undecumque serviatur.}
