\chapter{In Conceptione sanctae~Mariae Virginis}
\pslabel{form:conceptio}

\lelem{Antiphona super psalmos}
\lettrine{C}{um} iucunditate conceptionem sanctae Mariae celebremus:
ut ipsa pro nobis intercedat ad Dominum Iesum Christum.\quickref{can:conceptio_cumiucunditate}

\rubrica{Capitulum}
Trahe me post te.\quickref{assumptio:cap:trahe}
\rubrica{Hymnus}
Fit porta.\quickref{hym:fit_porta}
\elem{V.} Benedictus qui venit.\quickref{assumptio:verse:benedictus}

\lelem{Ad Nunc dimittis antiphona}
\lettrine{Q}{\editorial{uando}} concepta est Virgo sacratissima:
tunc illuminatus est mundus
stirps beata: radix sancta: et benedictus fructus eius.\quickref{can:conceptio_quandoconcepta}

\rubrica{Preces solitae:\quickref{preces}
  in fine versiculum adiungendo}\\ % "versiculum" here as full word, not abbreviated
Sentiant omnes tuum iuvamen.
\editorial{Quicumque celebrant tuam conceptionem.}\footcite[cf.][De BMV, f. 6v]{diu1523}\edissue{Try to find the verse in festal formularies of the main office, verify if I got it right.}

\lelem{Oratio}\pslabel{conceptio:oratio}
\lettrine{I}{mploramus} clementiam tuam Domine Deus noster:
ut qui festa Dei Genitricis colimus:
a cunctis malis imminentibus eius intercessionibus liberemur.
Per eundem.
