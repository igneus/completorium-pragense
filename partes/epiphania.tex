\formulary{In Epiphania Domini}
\label{form:epiphania}

\section{In Vigilia Epiphaniae}

\rubrica{Super psalmos antiphona
  \incipit{Verbum caro factum est.}\quickref{can:nativitas_verbumcaro}
  ut sero.
  Capitulum: hymnum: versum: et preces non dicimus.}

\lelem{Ad Nunc dimittis antiphona}
\lettrine{S}{tella} ista sicut flamma coruscat:
et regem regum demonstrat:
magi eam viderunt:
et Christo regi munera obtulerunt.\quickref{can:epiphania_stella}

\lelem{Oratio.}
\lettrine{D}{eus} illuminator omnium gentium:
da populis tuis perpetua pace gaudere:
et illud lumen splendidum infunde cordibus nostris:
quod trium magorum mentibus aspirasti.
Dominum nostrum Iesum Christum.

\rubrica{Non plus.}

\section{In die Epiphaniae}

\rubrica{Super psalmos antiphona
  \incipit{Verbum caro factum est.}\quickref{can:nativitas_verbumcaro}
  Capitulum: hymnus: versus
  ut in Nativitate Domini.\quickref{form:nativitas}
}\footnote{%
  \cite[116r]{bp1502}:
  \rubrica{Capitulum.}
  Omnes de Sabba venient aurum et thus deferentes:
  et laudem Domino annuntiantes.
  Deo gratias.
}

\rubrica{Ad Nunc dimittis antiphona
  \incipit{Stella ista.}\quickref{can:epiphania_stella}
  ut sero.
  Preces solitae\quickref{preces} dicuntur.
  In fine versum adiungendo.}\\
Venient ad te qui detrahebant tibi.
Et adorabunt vestigia pedum tuorum.\footnote{\cite[116r]{bp1502}: versus proprius deest}

\rubrica{Oratio \incipit{Deus illuminator gentium.}\quickref{form:epiphania}
  Haec sola.
  Hoc completorium durat usque Octavam Epiphaniae.}
