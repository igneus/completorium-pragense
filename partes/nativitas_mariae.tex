\section[In vigilia]{In Vigilia Nativitatis sanctae Mariae}

\lelem{Ad completorium super psalmos antiphona}
\lettrine{C}{um} iucunditate nativitatem \editorial{sanctae Mariae celebremus:
  ut ipsa pro nobis intercedat ad Dominum Iesum Christum}.\quickref{can:nativitas_mariae_cumiucunditate}

\rubrica{Capitulum}
Trahe me.\quickref{assumptio:cap:trahe}
\rubrica{Hymnus}
Fit porta.\quickref{hym:fit_porta}
\rubrica{V.}~Benedictus qui.\quickref{assumptio:verse:benedictus}

\lelem{Ad Nunc dimittis antiphona}
\lettrine{Q}{\editorial{uando}} nata est Virgo sacratissima:
tunc illuminatus est mundus
stirps beata: radix sancta: et benedictus fructus eius.\quickref{can:nativitas_mariae_quandonata}

\rubrica{Preces solitae\quickref{preces} in fine versiculum adiungendo}\\
Sentiant omnes
\editorial{tuum iuvamen. Quicumque celebrant tuam nativitatem.}\footnote{%
  cf. preces de Conceptione, p. \pageref{form:conceptio}}

\rubrica{Oratio}
Imploramus.\quickref{conceptio:oratio}
\rubrica{Haec sola.}
