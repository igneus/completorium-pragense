\chapter{In Vigilia Trinitatis}

\rubrica{Ad completorium psalmi consueti:\quickref{psalmi}
  quos sequitur.}
Alleluia.

\lelem{Capitulum}
\lettrine{G}{ratia} Domini nostri Iesu Christi
et caritas Dei et co\editorial{mmun}icatio Sancti Spiritus:
sit semper cum omnibus nobis. % "nobis" sic
Deo gratias.

\lelem{Hymnus}
\hymnus{laus_trinitati}

\elem{V.}
Custodi nos Domine ut pupillam oculi.\quickref{commune:versus:custodi}

\lelem{Ad Nunc dimittis antiphona}
\lettrine{L}{ibera} nos: salva nos: iustifica nos:
o beata Trinitas.\quickref{can:trinitas_libera}

\rubrica{Preces solitae\quickref{preces} dicuntur
  in fine nihil addendo.}

\lelem{Oratio}
\lettrine{F}{ilium} tuum omnipotens Pater
ante saecula benedictum invocantes oramus:
ut per Spiritum Sanctum humiliato calumniatore:
pacem nobis iustitiamque largiri dignetur.
Qui tecum vivit in unitate eiusdem.

\rubrica{Haec sola.
  Hic ordo completorii non mutatur
  usque Vigiliam Corporis Christi undecumque serviatur.}
