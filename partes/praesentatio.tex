\formulary{In festo Praesentationis sanctae Mariae}

\lelem{Ad completorium super psalmos antiphona}
\lettrine{C}{um} iucunditate praesentationem \editorial{sanctae Mariae celebremus:
  ut ipsa pro nobis intercedat ad Dominum Iesum Christum}.\quickref{can:praesentatio_cumiucunditate}

\rubrica{Capitulum}
Trahe me.\quickref{conceptio:cap:trahe}
\rubrica{Hymnus}
Fit porta.\quickref{hym:fit_porta}
\rubrica{V.}
Benedictus.\quickref{conceptio:verse:benedictus}
\rubrica{Ad Nunc dimittis antiphona}
Glorificamus.\quickref{circumcisio:an:glorificamus}

\rubrica{Preces solitae\quickref{preces} in fine versum adiungendo}\\
Sentiant omnes
\editorial{tuum iuvamen. Quicumque celebrant tuam praesentationem.}\footnote{%
  cf. preces de Conceptione, p. \pageref{form:conceptio}}
\rubrica{Oratio}
Imploramus.\quickref{conceptio:oratio}
