\chapter{In Nativitate Domini}
\pslabel{form:nativitas}

\rubrica{Ad completorium super psalmos consuetos\quickref{psalmi}
  antiphona}

\lettrine{V}{erbum} caro factum est: et habitavit\footnote{\cite[101r]{bp1502}: habitabit}
in nobis:
et vidimus gloriam eius gloriam quasi Unigeniti a Patre.\quickref{can:nativitas_verbumcaro}

\lelem{Capitulum}
\lettrine{P}{arvulus} natus est nobis: et filius datus est nobis.
Deo gratias.

\lelem{Hymnus}
\hymnus{corde}

\elem{V.} Benedictus qui venit\pslabel{nativitas:verse:benedictus}
\editorial{in nomine Domini. Deus Dominus et illuxit nobis.}%
  \footcite[De BMV, f. 6r]{diu1523}

\lelem{Ad Nunc dimittis antiphona}\pslabel{nativitas:magnum}
\lettrine{M}{agnum} nomen Domini Emanuel: quod annuntiatum est per Gabriel:
hodie apparuit in Israel:
per Mariam Virginem est natus\footnote{\cite[101r]{bp1502}: natus est}
Rex.\quickref{can:nativitas_magnum}

\rubrica{Preces consuetae.\quickref{preces}
  in fine versiculum istum adiungendo.}\\
Natus est nobis Salvator mundi.
Qui est Christus Dominus in civitate David.

\lelem{Oratio}\pslabel{nativitas:oratio}
\lettrine{P}{raesta} quaesumus omnipotens Deus:
ut natus hodie Salvator mundi:
sicut divinae nobis\footnote{\cite[101r]{bp1502}: -nobis}
generationis est auctor:
ita et immortalitatis sit ipse largitor.
Qui cum.

\rubrica{Haec sola.
  Hoc completorium non mutatur usque ad Circumcisionem Domini.
}\footnote{%
  \cite[101r]{bp1502}:
  \rubrica{usque ad Octavam Epiphaniae, praeter Vigiliam Circumcisionis
    et festum eius.}
}
