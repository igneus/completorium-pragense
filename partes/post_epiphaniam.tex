\chapter{Post Octavam Epiphaniae}

\rubrica{%
  Nota post Octavam Epiphaniae completorium commune\quickref{form:commune}
  dicitur usque ad dominicam primam Quadragesimae.
}

\rubrica{%
  Nota quod feria quarta in Capite ieiunii:
  et duobus sequentibus diebus:
  scilicet quinta et sexta feriis.
  completorium consuetum non mutamus.
  etiam si in eisdem feriis festa occurrant.
}
