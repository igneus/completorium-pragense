\chapter{In Annuntiatione sanctae Mariae}

\lelem{Ad completorium super psalmos antiphona}
\lettrine{C}{um} iucunditate annuntiationem sanctae Mariae
\editorial{celebremus:
  ut ipsa pro nobis intercedat ad Dominum Iesum Christum}.\edissue{notation}

\lelem{Capitulum}
\lettrine{B}{utyrum} et mel comedet:
ut sciat reprobare malum et eligere bonum.

\lelem{R\editorial{esponsorium}}
\lettrine{Q}{uo\editorial{modo}} fiet istud r\editorial{espo}ndens ait Maria:
q\editorial{uoniam} virum non cognosco.
angelus ad haec:
Spiritus Sanctus superveniet in te:
et virtus Altissimi obumbrabit tibi.\edissue{notation}

\rubrica{V.} Ideoque et quod nascetur ex te
sanctum vocabitur Filius Dei.
Spiritus.
Gloria.
Spiritus.

\rubrica{Hymnus.}
Fit porta.\quickref{hym:fit_porta}
\rubrica{V.} Benedictus qui.\quickref{conceptio:verse:benedictus}

\lelem{Ad Nunc dimittis antiphona}\pslabel{annuntiatio:an:alma}
\lettrine{A}{lma} Redemptoris Mater
quae pervia caeli porta manens:\edissue{manens sic, but compare other sources}
et stella maris:
succurre cadenti: surgere qui curat populo:
tu quae genuisti natura mirante
tuum sanctum Genitorem:
Virgo prius ac posterius:
Gabrielis ab ore sumens illud ave:
peccatorum miserere.\edissue{notation}

\rubrica{Preces solitae\quickref{preces} dicuntur.
  Versiculum in fine istum adiungendo.}\\
Ecce ancilla Domini.
Fiat mihi \editorial{secundum verbum tuum}.

\lelem{Oratio}
\lettrine{G}{ratiam} tuam quaesumus Domine
mentibus nostris infunde:
ut qui angelo annuntiante
Christi Filii tui incarnationem cognovimus:
per passionem eius et crucem
ad resurrectionis gloriam perducamur.
Qui tecum.

\rubrica{Haec sola dicitur.}

\vspace{2mm}

\rubrica{In crastino: hoc est ipso die
  Annuntiationis sanctae Mariae.
  Ad completorium omnia ut sero cum r\editorial{esponsorio}}
Quomodo fiet.
\rubrica{Ad Nunc dimittis vero ista antiphona dicitur.}

\lettrine{H}{aec} dies quam fecit Dominus.
Hodie Dominus afflictionem populi sui respexit
et redemptionem misit.
Hodie mortem quam femina intulit: femina fugavit.
Hodie Deus homo factus:
it quod fuit permansit: et quod non erat assumpsit.
Ergo exordium nostrae redemptionis devote recolamus:
et exsultemus dicentes.
Gloria tibi Domine.\edissue{notation}
